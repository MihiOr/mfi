cmake_minimum_required(VERSION 3.20)
project(mfi VERSION 0.1.0 LANGUAGES CXX)
option(MFI_BUILD_TESTS "Build host codec and transport tests" ON)
option(MFI_BUILD_EXAMPLES "Build the host example" ON)
option(MFI_BUILD_STM32F1 "Build the Cortex-M3 assembly backend" OFF)
add_library(mfi src/codec.cpp)
add_library(mfi::mfi ALIAS mfi)
target_include_directories(mfi PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(mfi PUBLIC cxx_std_14)
if(MFI_BUILD_STM32F1)
    add_library(mfi_stm32f1 ports/stm32f1/fast.cpp)
    target_link_libraries(mfi_stm32f1 PUBLIC mfi)
    target_include_directories(mfi_stm32f1 PUBLIC ports/stm32f1/include)
    foreach(name GPIO_BASE SIGNAL_RX_PIN DATA_TX_PIN DATA_RX_PIN SIGNAL_TX_PIN)
        if(DEFINED MFI_${name})
            target_compile_definitions(mfi_stm32f1 PUBLIC MFI_${name}=${MFI_${name}})
        endif()
    endforeach()
    add_library(mfi_stm32_example OBJECT examples/stm32f1.cpp)
    target_link_libraries(mfi_stm32_example PRIVATE mfi_stm32f1)
    add_executable(transport_elf tests/arm_anchor.cpp)
    target_link_libraries(transport_elf PRIVATE mfi_stm32f1)
    target_link_options(transport_elf PRIVATE -nostdlib -Wl,-e,mfi_send_asm -Wl,-Ttext=0x08000000)
    set_target_properties(transport_elf PROPERTIES OUTPUT_NAME transport SUFFIX .elf)
endif()
if(MFI_BUILD_EXAMPLES AND NOT CMAKE_CROSSCOMPILING)
    add_executable(mfi_example examples/codec.cpp)
    target_link_libraries(mfi_example PRIVATE mfi)
endif()
if(MFI_BUILD_TESTS AND NOT CMAKE_CROSSCOMPILING)
    enable_testing()
    foreach(name codec transport link)
        add_executable(${name}_tests tests/${name}_tests.cpp)
        target_link_libraries(${name}_tests PRIVATE mfi)
        if(MSVC)
            target_compile_options(${name}_tests PRIVATE /UNDEBUG)
        else()
            target_compile_options(${name}_tests PRIVATE -UNDEBUG -Wall -Wextra -Werror)
        endif()
        add_test(NAME ${name} COMMAND ${name}_tests)
    endforeach()
endif()
